Quick Reference Cheat Sheet
Every key pattern on one page. Read this before walking into the exam.

Binary Tree

;; BT is one of: empty | (list Nat BT BT) (define (mk-node l lf r) (list l lf r)) (define (get-label t) (first t)) (define (get-left t) (second t)) (define (get-right t) (third t)) ;; Template: (define (fn-bt t) (cond [(empty? t) ...] [else (... (get-label t) (fn-bt (get-left t)) (fn-bt (get-right t)))]))

BST Search

(define (bst-contains? n bst) (cond [(empty? bst) false] [(= (get-label bst) n) true] [(> (get-label bst) n) (bst-contains? n (get-left bst))] [else (bst-contains? n (get-right bst))])) ;; BST property: ;; label > ALL labels in left subtree ;; label < ALL labels in right subtree ;; O(log n) on balanced tree

NTree (General Tree)

;; NTree is one of: ;; Num OR ;; (cons NTree (listof NTree)) (define (fn-ntree nt) (cond [(number? nt) ...] ; leaf [(empty? (rest nt)) (... (fn-ntree (first nt)))] [else (... (fn-ntree (first nt)) (fn-ntree (rest nt)))]))

Lambda + Beta Reduction

;; Two equivalent definitions: (define (f x) (+ x 2)) (define f (lambda (x) (+ x 2))) ;; Beta-reduction: substitute x with arg ((lambda (x) (* x x)) 5) => (* 5 5) => 25 ;; Closure: lambda captures outer vars (define (make-adder n) (lambda (x) (+ x n))) (define add3 (make-adder 3)) (add3 10) => 13

Curry / Uncurry

(define (curry f) (lambda (x) (lambda (y) (f x y)))) (define (uncurry f) (lambda (x y) ((f x) y))) ;; Usage: (((curry +) 3) 4) ; => 7 (((curry cons) 0) '(1 2 3)) ; => '(0 1 2 3)

Accumulator Pattern

(define (rev lst) (local [(define (rev/acc lst acc) (cond [(empty? lst) acc] [else (rev/acc (rest lst) (cons (first lst) acc))]))] (rev/acc lst empty))) ; acc = "elements seen so far, reversed" ; Start with acc = empty ; Base case returns acc

filter / map

;; filter: (X->Bool) (listof X) -> (listof X) (filter even? '(1 2 3 4 5 6)) ; => '(2 4 6) ;; map: (X->Y) (listof X) -> (listof Y) (map sqr '(1 2 3 4)) ; => '(1 4 9 16) (map (lambda (x) (* x 2)) '(1 2 3)) ; => '(2 4 6)

foldr / foldl

;; foldr: RIGHT to LEFT ;; (foldr f b '(x1 x2 x3)) ;; = (f x1 (f x2 (f x3 b))) (foldr + 0 '(1 2 3)) ; => 6 (foldr cons '() '(1 2 3)) ; => '(1 2 3) ;; foldl: LEFT to RIGHT — reverses cons! ;; (foldl f b '(x1 x2 x3)) ;; = (f x3 (f x2 (f x1 b))) (foldl cons '() '(1 2 3)) ; => '(3 2 1)

build-list

;; build-list n f ;; = (list (f 0) (f 1) ... (f (n-1))) (build-list 4 identity) ; => '(0 1 2 3) (build-list 5 sqr) ; => '(0 1 4 9 16) (build-list 4 even?) ; => '(T F T F) (build-list 3 (lambda (i) (* i 2))) ; => '(0 2 4)

Local Expression

(local [(define x 5) (define (f n) (+ n x))] (f 3)) ; => 8 ; x and f are invisible OUTSIDE the local ;; 4 benefits: clarity, encapsulation, ;; scope-reuse, efficiency (compute once) (define (heron a b c) (local [(define s (/ (+ a b c) 2))] (sqrt (* s (- s a) (- s b) (- s c)))))

Graph + find-path

(define g '((A (B C)) (B (D)) (C ()) (D ()))) (define (neighbours nd g) (second (assoc nd g))) (define (find-path orig dest g) (cond [(symbol=? orig dest) (list dest)] [else (local [(define p (fp-list (neighbours orig g) dest g))] (cond [(empty? p) empty] [else (cons orig p)]))]))
